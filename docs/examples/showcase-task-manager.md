## 1. Source Project Overview

**TaskFlow** is a production FastAPI REST API for task and project management with real-time webhook notifications. Built on Python 3.12, SQLAlchemy 2.x, Alembic, and Pydantic v2, tested with pytest. It exposes ~12 endpoints across tasks, projects, comments, and webhook subscriptions, with ~15 Pydantic schemas, 4 enums, and 2 custom error types. After running `ADIK.orchestrator.md`, the repository gains a `sdk/` directory, a populated `Registry.md`, and updated context files — all shown below.

```typescript
import {
  CreateTaskRequest, TaskResponse, UpdateTaskRequest, ListTasksQuery,
  PaginatedResponse, ProjectResponse, CommentCreate, CommentResponse,
  WebhookSubscription, WebhookSubscriptionResponse,
} from './models';
import { TaskNotFoundError, DuplicateWebhookError } from './errors';

export interface TaskFlowClientOptions { baseUrl: string; accessToken?: string; }

export class TaskFlowClient {
  private baseUrl: string;
  private headers: Record<string, string>;

  constructor(opts: TaskFlowClientOptions) {
    this.baseUrl = opts.baseUrl.replace(/\/$/, '');
    this.headers = { 'Content-Type': 'application/json' };
    if (opts.accessToken) this.headers['Authorization'] = `Bearer ${opts.accessToken}`;
  }

  async createTask(dto: CreateTaskRequest): Promise<TaskResponse> {
    return this.post('/tasks', dto);
  }

  async getTask(taskId: string): Promise<TaskResponse> {
    return this.get(`/tasks/${taskId}`);
  }

  async updateTask(taskId: string, dto: UpdateTaskRequest): Promise<TaskResponse> {
    return this.put(`/tasks/${taskId}`, dto);
  }

  async deleteTask(taskId: string): Promise<void> {
    return this.delete(`/tasks/${taskId}`);
  }

  async subscribeWebhook(dto: WebhookSubscription): Promise<WebhookSubscriptionResponse> {
    return this.post('/webhooks/subscribe', dto);
  }

  // listTasks, listProjects, createComment, deleteWebhook follow the same pattern
  // — private HTTP helpers and error mapping omitted for brevity —
}
```

### `sdk/python/taskflow/client.py`

```python
from __future__ import annotations
import httpx
from .models import (CreateTaskRequest, TaskResponse, UpdateTaskRequest,
                     WebhookSubscription, WebhookSubscriptionResponse)
from .errors import TaskNotFoundError, DuplicateWebhookError

class TaskFlowClient:
    def __init__(self, base_url: str, *, access_token: str | None = None) -> None:
        headers: dict[str, str] = {}
        if access_token:
            headers["Authorization"] = f"Bearer {access_token}"
        self._client = httpx.AsyncClient(base_url=base_url, headers=headers)

    async def create_task(self, dto: CreateTaskRequest) -> TaskResponse:
        res = await self._client.post("/tasks", json=dto.to_dict())
        self._raise_for_status(res)
        return TaskResponse.from_dict(res.json())

    async def get_task(self, task_id: str) -> TaskResponse:
        res = await self._client.get(f"/tasks/{task_id}")
        self._raise_for_status(res)
        return TaskResponse.from_dict(res.json())

    async def subscribe_webhook(self, dto: WebhookSubscription) -> WebhookSubscriptionResponse:
        res = await self._client.post("/webhooks/subscribe", json=dto.to_dict())
        self._raise_for_status(res)
        return WebhookSubscriptionResponse.from_dict(res.json())

    # update_task, delete_task, list_tasks, list_projects, create_comment follow same pattern

    async def close(self) -> None:
        await self._client.aclose()

    async def __aenter__(self) -> TaskFlowClient:
        return self

    async def __aexit__(self, *exc: object) -> None:
        await self.close()

    # — _raise_for_status maps 404 → TaskNotFoundError, 409 → DuplicateWebhookError —
```

```csharp
namespace TaskFlow.Sdk;

public sealed class TaskFlowClient : IDisposable
{
    private readonly HttpClient _http;

    public TaskFlowClient(string baseUrl, string? accessToken = null)
    {
        _http = new HttpClient { BaseAddress = new Uri(baseUrl) };
        if (accessToken is not null)
            _http.DefaultRequestHeaders.Authorization =
                new("Bearer", accessToken);
    }

    public Task<TaskResponseDto> CreateTaskAsync(CreateTaskRequestDto dto) =>
        PostAsync<TaskResponseDto>("/tasks", dto);

    public Task<TaskResponseDto> GetTaskAsync(string taskId) =>
        GetAsync<TaskResponseDto>($"/tasks/{taskId}");

    public Task<WebhookSubscriptionResponseDto> SubscribeWebhookAsync(WebhookSubscriptionDto dto) =>
        PostAsync<WebhookSubscriptionResponseDto>("/webhooks/subscribe", dto);
```

## 7. Developer Experience

This is what consuming the SDK looks like after installation.

### TypeScript

```typescript
import { TaskFlowClient } from '@taskflow/sdk';
import { Priority, TaskStatus, WebhookEvent } from '@taskflow/sdk/enums';

const client = new TaskFlowClient({
  baseUrl: 'https://taskflow.example.com',
  accessToken: 'eyJhbGciOi...',
});

// Create a task — every field is typed, every enum autocompletes
const task = await client.createTask({
  title: 'Design landing page',
  projectId: 'proj_42',
  priority: Priority.High,
  labels: ['design', 'frontend'],
  dueDate: '2025-07-01',
});

console.log(task.status); // TaskStatus.Todo — enum, not a magic string

// Subscribe to real-time webhook events
const hook = await client.subscribeWebhook({
  url: 'https://hooks.example.com/taskflow',
  events: [WebhookEvent.TaskCompleted, WebhookEvent.TaskDeleted],
  secret: 'whsec_abc123',
});
```
