## 1. Source Project Overview

**PaymentsAPI** is a production NestJS REST API for payment processing built on Node.js, TypeORM, and PostgreSQL, tested with Jest. It has ~15 endpoints across auth, payments, refunds, and webhooks, with ~20 DTOs, 5 enums, and 3 custom error types.

After running `ADIK.orchestrator.md`, the repository gains a `sdk/` directory, a populated `Registry.md`, and updated context files. Everything below is what the pipeline produced.

### `sdk/typescript/src/client.ts`

```typescript
import {
  CreatePaymentDTO, PaymentResponseDTO, CreateRefundDTO,
  RefundResponseDTO, LoginDTO, RegisterDTO, AuthTokenDTO,
  ListPaymentsQueryDTO, PaginatedResponse,
} from './models';
import { PaymentDeclinedError, InsufficientFundsError } from './errors';

export interface PaymentsApiClientOptions {
  baseUrl: string;
  apiKey?: string;
  accessToken?: string;
}

export class PaymentsApiClient {
  private baseUrl: string;
  private headers: Record<string, string>;

  constructor(options: PaymentsApiClientOptions) {
    this.baseUrl = options.baseUrl.replace(/\/$/, '');
    this.headers = { 'Content-Type': 'application/json' };
    if (options.accessToken) this.headers['Authorization'] = `Bearer ${options.accessToken}`;
    if (options.apiKey) this.headers['X-API-Key'] = options.apiKey;
  }

  async createPayment(dto: CreatePaymentDTO): Promise<PaymentResponseDTO> {
    return this.post('/payments', dto);
  }

  async getPayment(id: string): Promise<PaymentResponseDTO> {
    return this.get(`/payments/${id}`);
  }

  async listPayments(query?: ListPaymentsQueryDTO): Promise<PaginatedResponse<PaymentResponseDTO>> {
    const qs = query ? '?' + new URLSearchParams(query as Record<string, string>).toString() : '';
    return this.get(`/payments${qs}`);
  }

  async cancelPayment(id: string): Promise<PaymentResponseDTO> {
    return this.post(`/payments/${id}/cancel`, {});
  }

  async createRefund(dto: CreateRefundDTO): Promise<RefundResponseDTO> {
    return this.post('/refunds', dto);
  }

  async login(dto: LoginDTO): Promise<AuthTokenDTO> {
    return this.post('/auth/login', dto);
  }

  async register(dto: RegisterDTO): Promise<AuthTokenDTO> {
    return this.post('/auth/register', dto);
  }

  // — private HTTP helpers and error mapping omitted for brevity —
}
```

### `sdk/python/payments_api/client.py`

```python
from __future__ import annotations
import httpx
from .models.create_payment_dto import CreatePaymentDTO
from .models.payment_response_dto import PaymentResponseDTO
from .models.create_refund_dto import CreateRefundDTO
from .models.refund_response_dto import RefundResponseDTO
from .models.auth_token_dto import AuthTokenDTO
from .errors import PaymentDeclinedError, InsufficientFundsError


class PaymentsApiClient:
    def __init__(self, base_url: str, *, access_token: str | None = None) -> None:
        headers: dict[str, str] = {}
        if access_token:
            headers["Authorization"] = f"Bearer {access_token}"
        self._client = httpx.AsyncClient(base_url=base_url, headers=headers)

    async def create_payment(self, dto: CreatePaymentDTO) -> PaymentResponseDTO:
        res = await self._client.post("/payments", json=dto.to_dict())
        self._raise_for_status(res)
        return PaymentResponseDTO.from_dict(res.json())

    async def get_payment(self, payment_id: str) -> PaymentResponseDTO:
        res = await self._client.get(f"/payments/{payment_id}")
        self._raise_for_status(res)
        return PaymentResponseDTO.from_dict(res.json())

    async def create_refund(self, dto: CreateRefundDTO) -> RefundResponseDTO:
        res = await self._client.post("/refunds", json=dto.to_dict())
        self._raise_for_status(res)
        return RefundResponseDTO.from_dict(res.json())

    async def login(self, email: str, password: str) -> AuthTokenDTO:
        res = await self._client.post("/auth/login", json={"email": email, "password": password})
        self._raise_for_status(res)
        return AuthTokenDTO.from_dict(res.json())

    async def close(self) -> None:
        await self._client.aclose()

    # — _raise_for_status maps 402 → PaymentDeclinedError / InsufficientFundsError —
```

using System.Net.Http.Json;
using PaymentsApi.Sdk.Models;
using PaymentsApi.Sdk.Exceptions;

namespace PaymentsApi.Sdk;

public sealed class PaymentsApiClient : IDisposable
{
    private readonly HttpClient _http;

    public PaymentsApiClient(string baseUrl, string? accessToken = null)
    {
        _http = new HttpClient { BaseAddress = new Uri(baseUrl) };
        if (accessToken is not null)
            _http.DefaultRequestHeaders.Authorization =
                new System.Net.Http.Headers.AuthenticationHeaderValue("Bearer", accessToken);
    }

    public Task<PaymentResponseDto> CreatePaymentAsync(CreatePaymentDto dto) =>
        PostAsync<PaymentResponseDto>("/payments", dto);

    public Task<PaymentResponseDto> GetPaymentAsync(string id) =>
        GetAsync<PaymentResponseDto>($"/payments/{id}");

    public Task<RefundResponseDto> CreateRefundAsync(CreateRefundDto dto) =>
        PostAsync<RefundResponseDto>("/refunds", dto);

    public Task<AuthTokenDto> LoginAsync(LoginDto dto) =>
        PostAsync<AuthTokenDto>("/auth/login", dto);

    // — private Get/Post helpers + error mapping omitted for brevity —

    public void Dispose() => _http.Dispose();
}

## 7. Developer Experience

This is what consuming the SDK looks like after `npm install @paymentsapi/sdk`.

### TypeScript

```typescript
import { PaymentsApiClient } from '@paymentsapi/sdk';
import { Currency, PaymentMethod, PaymentStatus } from '@paymentsapi/sdk/enums';

const client = new PaymentsApiClient({
  baseUrl: 'https://api.example.com',
  accessToken: 'eyJhbGciOi...',
});

// Create a payment — fully typed, autocomplete on every field
const payment = await client.createPayment({
  amount: 2500,
  currency: Currency.USD,
  method: PaymentMethod.Card,
  description: 'Order #1042',
});

console.log(payment.status); // PaymentStatus.Pending — enum, not a magic string

// Refund it
const refund = await client.createRefund({ paymentId: payment.id, amount: 2500 });
```
