# ADIK Post-Deployment Showcase — ChatHub (TypeScript Service)

> **What this guide covers:** A complete, real-world snapshot of what ADIK produces after a single run against a **TypeScript Chat Service** with both REST and WebSocket interfaces. Every artifact shown here was generated by the agentic pipeline with zero manual intervention.

## 1. Source Project Overview

**ChatHub** is a real-time chat service built on Express, Socket.io, Redis (pub/sub), and Prisma ORM, tested with Vitest. Unlike a pure REST API, ChatHub is a **hybrid service** — REST endpoints handle channel management, message history, and file uploads, while WebSocket events power real-time messaging, typing indicators, and presence tracking. It exposes ~10 REST endpoints + ~6 WebSocket events, with ~12 types, 3 enums, and 3 custom error types.

This is the first showcase where `projectType` is **"service"** rather than "api", and the registry includes a **WebSocket Events** section — demonstrating how ADIK handles non-REST protocols.

### `sdk/typescript/src/client.ts` — REST Client

```typescript
import {
  ChannelCreateDTO, ChannelResponse, SendMessageDTO, MessageResponse,
  MessageHistoryQuery, PaginatedResponse, ReactionDTO, PresenceUpdate, AttachmentMeta,
} from './models';
import { ChannelNotFoundError, MessageTooLargeError, RateLimitExceededError } from './errors';

export interface ChatHubClientOptions { baseUrl: string; accessToken?: string; }

export class ChatHubClient {
  private baseUrl: string;
  private headers: Record<string, string>;

  constructor(opts: ChatHubClientOptions) {
    this.baseUrl = opts.baseUrl.replace(/\/$/, '');
    this.headers = { 'Content-Type': 'application/json' };
    if (opts.accessToken) this.headers['Authorization'] = `Bearer ${opts.accessToken}`;
  }

  async listChannels(): Promise<ChannelResponse[]> { return this.get('/channels'); }
  async createChannel(dto: ChannelCreateDTO): Promise<ChannelResponse> { return this.post('/channels', dto); }
  async getMessages(channelId: string, q?: MessageHistoryQuery): Promise<PaginatedResponse<MessageResponse>> {
    const qs = q ? '?' + new URLSearchParams(q as Record<string, string>).toString() : '';
    return this.get(`/channels/${channelId}/messages${qs}`);
  }
  async sendMessage(channelId: string, dto: SendMessageDTO): Promise<MessageResponse> {
    return this.post(`/channels/${channelId}/messages`, dto);
  }
  async addReaction(messageId: string, dto: ReactionDTO): Promise<MessageResponse> {
    return this.put(`/messages/${messageId}/reactions`, dto);
  }
  async getPresence(userId: string): Promise<PresenceUpdate> { return this.get(`/users/${userId}/presence`); }
  async uploadAttachment(file: File | Blob, filename: string): Promise<AttachmentMeta> {
    const form = new FormData(); form.append('file', file, filename);
    return this.postForm('/attachments/upload', form);
  }

  // — private HTTP helpers and error mapping omitted for brevity —
}
```

### `sdk/typescript/src/realtime.ts` — WebSocket Client

```typescript
import { io, Socket } from 'socket.io-client';
import { SendMessageDTO, MessageResponse, PresenceUpdate } from './models';

export interface ChatHubRealtimeOptions { url: string; token: string; }
type MessageHandler = (msg: MessageResponse) => void;
type PresenceHandler = (update: PresenceUpdate) => void;

export class ChatHubRealtime {
  private socket: Socket;

  constructor(opts: ChatHubRealtimeOptions) {
    this.socket = io(opts.url, { auth: { token: opts.token }, transports: ['websocket'] });
  }

  sendMessage(dto: SendMessageDTO): void { this.socket.emit('message:send', dto); }
  onMessage(handler: MessageHandler): void { this.socket.on('message:receive', handler); }
  startTyping(channelId: string): void { this.socket.emit('typing:start', { channelId }); }
  stopTyping(channelId: string): void { this.socket.emit('typing:stop', { channelId }); }
  onPresence(handler: PresenceHandler): void { this.socket.on('presence:update', handler); }
  disconnect(): void { this.socket.disconnect(); }
}
```

### `sdk/python/chathub/client.py` — REST Client

```python
from __future__ import annotations
import httpx
from .models import (ChannelCreateDTO, ChannelResponse, SendMessageDTO,
                     MessageResponse, PresenceUpdate, AttachmentMeta)
from .errors import ChannelNotFoundError, MessageTooLargeError, RateLimitExceededError

class ChatHubClient:
    def __init__(self, base_url: str, *, access_token: str | None = None) -> None:
        headers: dict[str, str] = {}
        if access_token:
            headers["Authorization"] = f"Bearer {access_token}"
        self._client = httpx.AsyncClient(base_url=base_url, headers=headers)

    async def list_channels(self) -> list[ChannelResponse]:
        res = await self._client.get("/channels"); self._raise_for_status(res)
        return [ChannelResponse.from_dict(c) for c in res.json()]

    async def send_message(self, channel_id: str, dto: SendMessageDTO) -> MessageResponse:
        res = await self._client.post(f"/channels/{channel_id}/messages", json=dto.to_dict())
        self._raise_for_status(res); return MessageResponse.from_dict(res.json())

    async def close(self) -> None: await self._client.aclose()
    # — create_channel, get_messages, add_reaction, get_presence, upload_attachment follow same pattern —
    # — _raise_for_status maps 404 → ChannelNotFoundError, 413 → MessageTooLargeError, 429 → RateLimitExceededError —
```

### `sdk/python/chathub/realtime.py` — Async WebSocket Client

```python
from __future__ import annotations
import socketio
from .models import SendMessageDTO, MessageResponse, PresenceUpdate

class ChatHubRealtime:
    def __init__(self, url: str, token: str) -> None:
        self._sio = socketio.AsyncClient(reconnection=True)
        self._url, self._token = url, token

    async def connect(self) -> None:
        await self._sio.connect(self._url, auth={"token": self._token}, transports=["websocket"])

    def send_message(self, dto: SendMessageDTO) -> None:
        self._sio.emit("message:send", dto.to_dict())

    def on_message(self, handler) -> None:
        @self._sio.on("message:receive")
        async def _wrapper(data): await handler(MessageResponse.from_dict(data))

    def on_presence(self, handler) -> None:
        @self._sio.on("presence:update")
        async def _wrapper(data): await handler(PresenceUpdate.from_dict(data))

    async def disconnect(self) -> None: await self._sio.disconnect()
```

```csharp
using System.Net.Http.Json;
using ChatHub.Sdk.Models;
using ChatHub.Sdk.Exceptions;

namespace ChatHub.Sdk;

public sealed class ChatHubClient : IDisposable
{
    private readonly HttpClient _http;

    public ChatHubClient(string baseUrl, string? accessToken = null)
    {
        _http = new HttpClient { BaseAddress = new Uri(baseUrl) };
        if (accessToken is not null)
            _http.DefaultRequestHeaders.Authorization = new("Bearer", accessToken);
    }

    public Task<ChannelResponseDto[]> ListChannelsAsync() => GetAsync<ChannelResponseDto[]>("/channels");
    public Task<ChannelResponseDto> CreateChannelAsync(ChannelCreateDto dto) => PostAsync<ChannelResponseDto>("/channels", dto);
    public Task<MessageResponseDto> SendMessageAsync(string channelId, SendMessageDto dto) =>
        PostAsync<MessageResponseDto>($"/channels/{channelId}/messages", dto);
    // GetMessagesAsync, AddReactionAsync, GetPresenceAsync, UploadAttachmentAsync follow same pattern

    public void Dispose() => _http.Dispose();
}
```

### `sdk/csharp/ChatHub.Sdk/ChatHubRealtime.cs` — SignalR-Style Real-Time Client

```csharp
using Microsoft.AspNetCore.SignalR.Client;
using ChatHub.Sdk.Models;

namespace ChatHub.Sdk;

public sealed class ChatHubRealtime : IAsyncDisposable
{
    private readonly HubConnection _hub;

    public ChatHubRealtime(string url, string token)
    {
        _hub = new HubConnectionBuilder()
            .WithUrl(url, opts => opts.AccessTokenProvider = () => Task.FromResult<string?>(token))
            .WithAutomaticReconnect().Build();
    }

    public Task ConnectAsync() => _hub.StartAsync();
    public Task SendMessageAsync(SendMessageDto dto) => _hub.InvokeAsync("message:send", dto);
    public void OnMessage(Action<MessageResponseDto> handler) => _hub.On("message:receive", handler);
    public void OnPresence(Action<PresenceUpdateDto> handler) => _hub.On("presence:update", handler);
    public ValueTask DisposeAsync() => _hub.DisposeAsync();
}
```

## 7. Developer Experience

This is what consuming the SDK looks like — both REST operations and real-time messaging.

### TypeScript

```typescript
import { ChatHubClient, ChatHubRealtime } from '@chathub/sdk';
import { MessageType, ChannelType } from '@chathub/sdk/enums';

// REST: create a channel
const rest = new ChatHubClient({
  baseUrl: 'https://chat.example.com',
  accessToken: 'eyJhbGciOi...',
});

const channel = await rest.createChannel({
  name: 'engineering',
  type: ChannelType.Public,
  memberIds: ['user_1', 'user_2'],
});

// Real-time: connect, listen, and send
const ws = new ChatHubRealtime({
  url: 'wss://chat.example.com',
  token: 'eyJhbGciOi...',
});

ws.onMessage((msg) => console.log(`${msg.authorId}: ${msg.content}`));
ws.onPresence((p) => console.log(`${p.userId} is ${p.status}`));

ws.sendMessage({ channelId: channel.id, content: 'Hello, team!' });
```

### Python

```python
from chathub import ChatHubClient
from chathub.realtime import ChatHubRealtime
from chathub.models import ChannelCreateDTO, SendMessageDTO
from chathub.enums import ChannelType

# REST
async with ChatHubClient("https://chat.example.com", access_token="eyJ...") as rest:
    channel = await rest.create_channel(ChannelCreateDTO(
        name="engineering", type=ChannelType.PUBLIC, member_ids=["user_1", "user_2"],
    ))

# Real-time
rt = ChatHubRealtime("wss://chat.example.com", token="eyJ...")
rt.on_message(lambda msg: print(f"{msg.author_id}: {msg.content}"))
await rt.connect()
rt.send_message(SendMessageDTO(channel_id=channel.id, content="Hello, team!"))
```

### C\#


Two client classes. REST for CRUD, WebSocket for real-time — both fully typed, both generated from the same registry. When ChatHub adds a new event, re-run `ADIK.orchestrator.md` and every SDK updates automatically.
